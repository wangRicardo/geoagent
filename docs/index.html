my_tool.py — 你的第一个领域工具 
from geoagent import registry

@registry.register(category="geophysics")
def nmo_velocity(t0_ms: float, vrms_ms: float, offset_m: float) -> str:
    """由 T0 与均方根速度计算给定偏移距的 NMO 时差。"""
    t = (t0_ms**2 + (offset_m / vrms_ms)**2 * 1e6) ** 0.5
    return f"NMO 时差: {t - t0_ms:.2f} ms @ offset={offset_m} m"

# 保存后，LLM 立刻就能调用 nmo_velocity —— 零胶水代码
